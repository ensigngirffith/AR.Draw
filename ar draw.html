    <script>

        const imageLoader = document.getElementById('image-loader');

        const opacitySlider = document.getElementById('opacity-slider');

        const sizeSlider = document.getElementById('size-slider');

        const lockBtn = document.getElementById('lock-btn');

        const drawingPlane = document.getElementById('drawing-plane');

        const marker = document.getElementById('marker');

        

        let isLocked = false;

        let savedPosition = { x: 0, y: 0, z: 0 };

        let savedRotation = { x: -90, y: 0, z: 0 };


        // 1. Handle Custom Image Uploads

        imageLoader.addEventListener('change', function(e) {

            const reader = new FileReader();

            reader.onload = function(event){

                // Update asset source dynamically

                document.getElementById('trace-image').setAttribute('src', event.target.result);

                // Refresh material texture map assignment

                drawingPlane.setAttribute('src', '');

                drawingPlane.setAttribute('src', '#trace-image');

            }

            if(e.target.files[0]) {

                reader.readAsDataURL(e.target.files[0]);

            }

        });


        // 2. Adjust Opacity Dynamically

        opacitySlider.addEventListener('input', function(e) {

            drawingPlane.setAttribute('material', `transparent: true; opacity: ${e.target.value}; shader: flat;`);

        });


        // 3. Adjust Image Plane Dimensions

        sizeSlider.addEventListener('input', function(e) {

            const scale = e.target.value;

            drawingPlane.setAttribute('scale', `${scale} ${scale} 1`);

        });


        // 4. Screen Lock Mode 

        // Locks the element relative to screen space if tracking drops out

        lockBtn.addEventListener('click', function() {

            isLocked = !isLocked;

            if(isLocked) {

                lockBtn.innerText = "Unlock Tracking";

                lockBtn.style.backgroundColor = "#FF3B30";

                

                // Detach from AR marker space and temporarily append directly onto scene camera perspective view

                const cameraEl = document.querySelector('[camera]');

                cameraEl.appendChild(drawingPlane);

                

                // Position fixedly right in front of camera lens view

                drawingPlane.setAttribute('position', '0 0 -1.5');

                drawingPlane.setAttribute('rotation', '0 0 0');

            } else {

                lockBtn.innerText = "Lock Tracking Screen";

                lockBtn.style.backgroundColor = "#007AFF";

                

                // Re-attach structure down back to AR target track node

                marker.appendChild(drawingPlane);

                drawingPlane.setAttribute('position', '0 0 0');

                drawingPlane.setAttribute('rotation', '-90 0 0');

            }

        });

    </script>
